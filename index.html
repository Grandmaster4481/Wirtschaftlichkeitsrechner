/* STREAMING_CHUNK:Configuring Google Apps Script backend... */ function doGet(e) { try { var ss = SpreadsheetApp.getActiveSpreadsheet(); // 1. Artikel auslesen (Reiter "Tabellenblatt1") var itemSheet = ss.getSheetByName("Tabellenblatt1") || ss.getSheets()[0]; var itemData = itemSheet.getDataRange().getValues(); var items = []; for (var i = 1; i < itemData.length; i++) { var row = itemData[i]; var name = String(row[0] || "").trim(); if (name !== "") { items.push({ name: name, market: String(row[1] || "Allgemein").trim(), quantity: String(row[2] || "1").trim(), checked: row[3] === true || String(row[3]).toLowerCase() === "true", id: row[4] ? String(row[4]) : 'id_' + i + '_' + Date.now() }); } } // 2. Märkte & Farben auslesen (Reiter "Markt") var marketSheet = ss.getSheetByName("Markt"); var markets = []; if (marketSheet) { var marketData = marketSheet.getDataRange().getValues(); for (var j = 0; j < marketData.length; j++) { var mRow = marketData[j]; var mName = String(mRow[0] || "").trim(); if (mName !== "" && mName.toLowerCase() !== "markt" && mName.toLowerCase() !== "name") { markets.push({ name: mName, bgColor: String(mRow[1] || "grau").trim(), textColor: String(mRow[2] || "weiss").trim() }); } } } if (markets.length === 0) { markets = [ { name: "Allgemein", bgColor: "grau", textColor: "weiss" }, { name: "Edeka", bgColor: "gelb", textColor: "blau" }, { name: "Aldi", bgColor: "grau", textColor: "hellblau" }, { name: "dm", bgColor: "grau", textColor: "blau" } ]; } return ContentService.createTextOutput(JSON.stringify({ items: items, markets: markets })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ error: err.toString(), items: [], markets: [] })) .setMimeType(ContentService.MimeType.JSON); } } function doPost(e) { try { var data = JSON.parse(e.postData.contents); var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheetByName("Tabellenblatt1") || ss.getSheets()[0]; if (sheet.getLastRow() === 0) { sheet.appendRow(["Artikel", "Markt", "Anzahl", "Erledigt", "ID"]); } var rows = sheet.getDataRange().getValues(); if (data.action === "add") { sheet.appendRow([data.name, data.market || "Allgemein", data.quantity || "1", false, data.id]); } else if (data.action === "toggle") { for (var i = 1; i < rows.length; i++) { if (String(rows[i][4]) === String(data.id)) { var currentVal = rows[i][3] === true || String(rows[i][3]).toLowerCase() === "true"; sheet.getRange(i + 1, 4).setValue(!currentVal); break; } } } else if (data.action === "delete") { for (var k = rows.length - 1; k >= 1; k--) { if (String(rows[k][4]) === String(data.id)) { sheet.deleteRow(k + 1); break; } } } return ContentService.createTextOutput(JSON.stringify({ success: true })) .setMimeType(ContentService.MimeType.JSON); } catch (err) { return ContentService.createTextOutput(JSON.stringify({ success: false, error: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } }
